// ===== ADMIN STATE MANAGEMENT ===== const adminState = { admin: null, settings: {}, isLoading: false, }; /* ========================================================= SETTINGS BRIDGE — sync admin settings to public pages ========================================================= */ function bridgeSettingsToUI() { // Bridge settings to public pages — same pattern as js/site-content.js document.querySelectorAll('[data-content], [data-content-html], [data-content-href]').forEach(el => { const key = el.dataset.content || el.dataset.contentHtml || el.dataset.contentHref || el.id; const content = adminState.settings[key]; if (content !== undefined) { if (el.dataset.content) { el.textContent = content; el.style.display = 'block'; } else if (el.dataset.contentHtml) { el.innerHTML = content; } else if (el.dataset.contentHref) { el.href = content; } } }); } /* ========================================================= SETTINGS LOADER — fetch from backend ========================================================= */ async function loadSettings() { if (adminState.isLoading) return; adminState.isLoading = true; try { const res = await fetch(API + '/settings'); if (!res.ok) throw new Error('Settings request failed'); const data = await res.json(); if (data.success && data.settings) { adminState.settings = data.settings; bridgeSettingsToUI(); } } catch(e) { console.error('Settings load error:', e); } finally { adminState.isLoading = false; } } /* ========================================================= SETTINGS SAVER — PUT to backend ========================================================= */ async function saveSettings() { if (adminState.isLoading) return; const settings = {}; document.querySelectorAll('[id^="set_"]').forEach(el => { const key = el.id.replace('set_', ''); if (el.type === 'checkbox') { settings[key] = el.checked; } else { settings[key] = el.value; } }); adminState.isLoading = true; const btn = document.querySelector('.settings-panel .save-btn'); const originalText = btn.textContent; btn.textContent = 'Saving...'; btn.disabled = true; try { const res = await fetch(API + '/settings', { method: 'PUT', credentials: 'include', headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + adminState.token }, body: JSON.stringify(settings) }); const data = await res.json(); if (data.success) { adminState.settings = settings; bridgeSettingsToUI(); alert('Settings saved!'); } else { alert('Save failed: ' + (data.message || 'Unknown error')); } } catch(e) { alert('Network error saving settings'); console.error(e); } finally { btn.textContent = originalText; btn.disabled = false; adminState.isLoading = false; } } /* ========================================================= UI BRIDGE — form population ========================================================= */ function populateSettingsForm() { document.querySelectorAll('[id^="set_"]').forEach(el => { const key = el.id.replace('set_', ''); if (adminState.settings[key] !== undefined) { if (el.type === 'checkbox') { el.checked = Boolean(adminState.settings[key]); } else { el.value = adminState.settings[key]; } } }); } /* ========================================================= TESTIMONIALS UI — bridge for testimonials settings ========================================================= */ function bridgeTestimonialsUI() { if (!adminState.settings.testimonialText || !adminState.settings.testimonialAuthor) return; const placeholder = document.querySelector('.testimonial-placeholder'); if (placeholder) { placeholder.textContent = adminState.settings.testimonialText; const author = document.querySelector('.testimonial-author'); if (author) author.textContent = "— " + adminState.settings.testimonialAuthor; } } /* ========================================================= INIT ADMIN APP ========================================================= */ async function initAdminApp() { try { const res = await fetch(API + '/me', { credentials: 'include' }); if (res.ok) { const data = await res.json(); if (data.success && data.admin) { adminState.admin = data.admin; adminState.token = data.token; await loadSettings(); populateSettingsForm(); bridgeTestimonialsUI(); bridgeSettingsToUI(); showApp(); return; } } } catch(e) { } showLogin(); } /* ========================================================= TOAST MESSENGER ========================================================= */ function toast(msg, type = 'success') { const container = document.getElementById('toast-container') || createToastContainer(); const toast = document.createElement('div'); toast.className = `toast ${type}`; toast.textContent = msg; container.appendChild(toast); setTimeout(() => toast.remove(), 3000); } function createToastContainer() { const container = document.createElement('div'); container.id = 'toast-container'; container.style.position = 'fixed'; container.style.top = '20px'; container.style.right = '20px'; container.style.zIndex = '9999'; container.style.display = 'flex'; container.style.flexDirection = 'column'; container.style.gap = '10px'; document.body.appendChild(container); return container; } /* ========================================================= MOBILE NAVIGATION ========================================================= */ function toggleSidebar() { const sidebar = document.getElementById('sidebar'); const overlay = document.getElementById('sidebarOverlay'); sidebar.classList.toggle('open'); overlay.classList.toggle('active'); document.body.style.overflow = sidebar.classList.contains('open') ? 'hidden' : ''; } function closeSidebar() { const sidebar = document.getElementById('sidebar'); const overlay = document.getElementById('sidebarOverlay'); sidebar.classList.remove('open'); overlay.classList.remove('active'); document.body.style.overflow = ''; } function handleOutsideClick(e) { const sidebar = document.getElementById('sidebar'); const toggleBtn = document.querySelector('.sidebar-toggle'); if (!sidebar.contains(e.target) && !toggleBtn.contains(e.target)) { closeSidebar(); } } function handleResize() { if (window.innerWidth > 1024) { closeSidebar(); } } /* ========================================================= CREATE MISSING UI ELEMENTS ========================================================= */ function ensureToastContainer() { if (!document.getElementById('toast-container')) { createToastContainer(); } } /* ========================================================= MAKE FUNCTIONS GLOBAL ========================================================= */ window.adminState = adminState; window.loadSettings = loadSettings; window.saveSettings = saveSettings; window.populateSettingsForm = populateSettingsForm; window.bridgeSettingsToUI = bridgeSettingsToUI; window.bridgeTestimonialsUI = bridgeTestimonialsUI; window.initAdminApp = initAdminApp; window.toast = toast; window.toggleSidebar = toggleSidebar; window.closeSidebar = closeSidebar; /* ========================================================= EVENT LISTENERS ========================================================= */ document.addEventListener('DOMContentLoaded', function() { ensureToastContainer(); initAdminApp(); document.querySelector('.sidebar-toggle')?.addEventListener('click', toggleSidebar); document.querySelector('#sidebarOverlay')?.addEventListener('click', closeSidebar); document.addEventListener('click', handleOutsideClick);
